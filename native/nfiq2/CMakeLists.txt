cmake_minimum_required(VERSION 3.20)
project(openphysical_nfiq2 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN;@loader_path")
set(OP_SANITIZER "" CACHE STRING "address or undefined")
if(OP_SANITIZER)
 add_compile_options(-fsanitize=${OP_SANITIZER} -fno-omit-frame-pointer)
 add_link_options(-fsanitize=${OP_SANITIZER})
endif()
set(U "${CMAKE_CURRENT_SOURCE_DIR}/upstream")
# Compile the pinned numerical dependencies; system OpenCV may produce other scores.
set(BUILD_LIST core,ml,imgproc CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
foreach(FLAG BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps BUILD_JAVA BUILD_opencv_python3 WITH_FFMPEG WITH_GTK WITH_OPENCL WITH_IPP WITH_ITT WITH_TBB WITH_OPENMP ENABLE_FAST_MATH WITH_ADE WITH_QUIRC WITH_OBSENSOR WITH_OPENJPEG WITH_TIFF WITH_PNG WITH_JPEG WITH_WEBP WITH_OPENEXR WITH_PROTOBUF)
 set(${FLAG} OFF CACHE BOOL "" FORCE)
endforeach()
set(OPENCV_GENERATE_PKGCONFIG OFF CACHE BOOL "" FORCE)
add_subdirectory("${U}/opencv" "${CMAKE_CURRENT_BINARY_DIR}/opencv" EXCLUDE_FROM_ALL)
# Keep typed statistical kernels behind signature-correct generic dispatch.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(STATISTICS_OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/opencv-statistics")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare-opencv.py"
 --upstream "${U}/opencv" --destination "${STATISTICS_OVERLAY}"
 --opencv-build "${CMAKE_CURRENT_BINARY_DIR}/opencv"
 COMMAND_ERROR_IS_FATAL ANY)
get_target_property(CORE_SOURCES opencv_core SOURCES)
list(FILTER CORE_SOURCES EXCLUDE REGEX "/((mean|sum|merge|split)\\.dispatch|arithm|mathfuncs|dxt|norm|minmax)\\.cpp$")
set_property(TARGET opencv_core PROPERTY SOURCES "${CORE_SOURCES};${STATISTICS_OVERLAY}/mean.dispatch.cpp;${STATISTICS_OVERLAY}/sum.dispatch.cpp;${STATISTICS_OVERLAY}/arithm.cpp;${STATISTICS_OVERLAY}/mathfuncs.cpp;${STATISTICS_OVERLAY}/merge.dispatch.cpp;${STATISTICS_OVERLAY}/split.dispatch.cpp;${STATISTICS_OVERLAY}/dxt.cpp;${STATISTICS_OVERLAY}/norm.cpp;${STATISTICS_OVERLAY}/minmax.cpp")
target_include_directories(opencv_core PRIVATE "${U}/opencv/modules/core/src")
# GCC's ARM signed-vector intrinsic implementation is instrumented as signed
# overflow even for OpenCV's explicitly wrapping operation. Perform that lane
# addition through unsigned vectors, then preserve the resulting signed bits.
file(READ "${U}/opencv/modules/core/include/opencv2/core/hal/intrin_neon.hpp" NEON_HEADER)
set(NEON_OLD "OPENCV_HAL_IMPL_NEON_BIN_FUNC(v_int16x8, v_add_wrap, vaddq_s16)")
string(FIND "${NEON_HEADER}" "${NEON_OLD}" NEON_POSITION)
if(NEON_POSITION LESS 0)
 message(FATAL_ERROR "Unexpected OpenCV NEON signed wrapping addition source")
endif()
string(REPLACE "${NEON_OLD}" "inline v_int16x8 v_add_wrap(const v_int16x8& a, const v_int16x8& b) { return v_int16x8(vreinterpretq_s16_u16(vaddq_u16(vreinterpretq_u16_s16(a.val), vreinterpretq_u16_s16(b.val)))); }" NEON_HEADER "${NEON_HEADER}")
set(OPENCV_OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/opencv-overlay")
file(MAKE_DIRECTORY "${OPENCV_OVERLAY}/opencv2/core/hal")
file(WRITE "${OPENCV_OVERLAY}/opencv2/core/hal/intrin_neon.hpp" "${NEON_HEADER}")
foreach(TARGET opencv_core opencv_imgproc opencv_ml)
 target_include_directories(${TARGET} BEFORE PRIVATE "${OPENCV_OVERLAY}")
endforeach()
# Avoid constructing an out-of-bounds row pointer for a masked image-border
# neighbor. Such rows cannot contribute to flood fill and are skipped directly.
file(READ "${U}/opencv/modules/imgproc/src/floodfill.cpp" FLOOD)
set(FLOOD_OLD "dir = data[k][0];
            img = (_Tp*)(pImage + (YC + dir) * step);")
string(FIND "${FLOOD}" "${FLOOD_OLD}" FLOOD_POSITION)
if(FLOOD_POSITION LESS 0)
 message(FATAL_ERROR "Unexpected OpenCV flood-fill neighbor source")
endif()
string(REPLACE "${FLOOD_OLD}" "dir = data[k][0];
            if (YC + dir < 0 || YC + dir >= image.rows) continue;
            img = (_Tp*)(pImage + (YC + dir) * step);" FLOOD "${FLOOD}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/floodfill.cpp" "${FLOOD}")
get_target_property(IMGPROC_SOURCES opencv_imgproc SOURCES)
list(FILTER IMGPROC_SOURCES EXCLUDE REGEX "/floodfill.cpp$")
set_property(TARGET opencv_imgproc PROPERTY SOURCES "${IMGPROC_SOURCES};${CMAKE_CURRENT_BINARY_DIR}/floodfill.cpp")
target_include_directories(opencv_imgproc PRIVATE "${U}/opencv/modules/imgproc/src")
set(FRFXLL_PATH "${U}/fingerjetfxose/FingerJetFXOSE/libFRFXLL")
# The third-order orientation term is weighted by constant zero upstream, yet its
# unused intermediates overflow signed integers. Drop only that dead calculation
# in a generated source copy; the active first/second-order terms are unchanged.
file(COPY "${FRFXLL_PATH}/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/fingerjet")
set(FRFXLL_PATH "${CMAKE_CURRENT_BINARY_DIR}/fingerjet")
file(READ "${FRFXLL_PATH}/src/algorithm/orimap.h" ORIMAP)
set(DEAD_TERM "complex<int32> Z3(5*(X30*X30 - X33*X33) + 3*(X31*X31 - X32*X32) + 2*(X30*X32 - X31*X33),
                              10*(X32*X33 + X30*X31) + 18*X31*X32 + 2*X30*X33);")
string(FIND "${ORIMAP}" "${DEAD_TERM}" DEAD_POSITION)
if(DEAD_POSITION LESS 0 OR NOT ORIMAP MATCHES "const int32 k3 = 0;")
 message(FATAL_ERROR "Unexpected FingerJet inactive orientation term")
endif()
string(REPLACE "${DEAD_TERM}" "complex<int32> Z3(0, 0);" ORIMAP "${ORIMAP}")
file(WRITE "${FRFXLL_PATH}/src/algorithm/orimap.h" "${ORIMAP}")
file(READ "${FRFXLL_PATH}/src/algorithm/extract_minutia.h" EXTRACT)
string(FIND "${EXTRACT}" "p[-3*width]" OFFSET_POSITION)
if(OFFSET_POSITION LESS 0)
 message(FATAL_ERROR "Unexpected FingerJet neighborhood offset")
endif()
string(REPLACE "p[-3*width]" "p[-3*int32(width)]" EXTRACT "${EXTRACT}")
file(WRITE "${FRFXLL_PATH}/src/algorithm/extract_minutia.h" "${EXTRACT}")


# Build FingerJet as a distinct, replaceable shared library.
file(GLOB FJ_SOURCES "${FRFXLL_PATH}/src/lib/*.cpp")
add_library(FRFXLL SHARED ${FJ_SOURCES})
# FingerJet fixed-point FFT shifts signed two's-complement values. C++20 defines
# those shifts modulo the width, matching the intended arithmetic without UB.
set_target_properties(FRFXLL PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_include_directories(FRFXLL PRIVATE "${FRFXLL_PATH}/include" "${FRFXLL_PATH}/src/algorithm")
set(A "${U}/NFIQ2/NFIQ2Algorithm")
file(GLOB NFIQ_SOURCES "${A}/src/nfiq2/*.cpp" "${A}/src/quality_modules/*.cpp" "${A}/src/prediction/*.cpp")
# Base64 decoding accumulates bits modulo the accumulator width. The original
# signed shift overflows while reading the embedded model. Generate an unsigned
# accumulator adaptation and keep the imported upstream file byte-identical.
file(READ "${A}/src/nfiq2/nfiq2_data.cpp" DATA_SOURCE)
string(REGEX MATCHALL "int bits = 0, data = 0" DATA_MATCHES "${DATA_SOURCE}")
list(LENGTH DATA_MATCHES DATA_MATCH_COUNT)
if(NOT DATA_MATCH_COUNT EQUAL 1)
 message(FATAL_ERROR "Unexpected NFIQ2 Base64 accumulator source")
endif()
string(REPLACE "int bits = 0, data = 0;" "int bits = 0; unsigned int data = 0;" DATA_SOURCE "${DATA_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nfiq2_data.cpp" "${DATA_SOURCE}")
list(FILTER NFIQ_SOURCES EXCLUDE REGEX "/nfiq2_data.cpp$")
list(APPEND NFIQ_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/nfiq2_data.cpp")
set(NFIQ2_VERSION_MAJOR 2)
set(NFIQ2_VERSION_MINOR 3)
set(NFIQ2_VERSION_PATCH 0)
set(NFIQ2_VERSION_FULL "2.3.0")
set(NFIQ2_VERSION "2.3.0")
set(NFIQ2_VERSION_STATUS "")
set(NFIQ2_BUILD_DATE "202410010000")
set(NFIQ2_VERSION_COMMIT "76b8c4e0b0541f3deab832b1a496e524edc0b5b6")
configure_file("${A}/src/nfiq2/version.cpp.in" "${CMAKE_CURRENT_BINARY_DIR}/version.cpp" @ONLY)
# Test builds inject a real allocation exception inside the upstream catch
# boundary. Release sources retain the original computation byte for byte.
include(CTest)
if(BUILD_TESTING)
 file(READ "${A}/src/nfiq2/nfiq2_algorithm_impl.cpp" ALGORITHM_SOURCE)
 string(REGEX MATCHALL "modules = NFIQ2::QualityMeasures::" FEATURE_ASSIGNMENTS "${ALGORITHM_SOURCE}")
 list(LENGTH FEATURE_ASSIGNMENTS FEATURE_ASSIGNMENT_COUNT)
 if(NOT FEATURE_ASSIGNMENT_COUNT EQUAL 1)
  message(FATAL_ERROR "Unexpected NFIQ2 feature-computation injection point")
 endif()
 string(REPLACE "modules = NFIQ2::QualityMeasures::" "op_nfiq2_test_feature_allocation();\n\t\tmodules = NFIQ2::QualityMeasures::" ALGORITHM_SOURCE "${ALGORITHM_SOURCE}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nfiq2_algorithm_impl.cpp" "extern \"C\" void op_nfiq2_test_feature_allocation();\n${ALGORITHM_SOURCE}")
 list(FILTER NFIQ_SOURCES EXCLUDE REGEX "/nfiq2_algorithm_impl.cpp$")
 list(APPEND NFIQ_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/nfiq2_algorithm_impl.cpp")
endif()
add_library(openphysical_nfiq2 SHARED op_nfiq2.cpp ${NFIQ_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/version.cpp")
target_compile_definitions(openphysical_nfiq2 PRIVATE NFIQ2_EMBED_RANDOM_FOREST_PARAMETERS NFIQ2_EMBEDDED_RANDOM_FOREST_PARAMETERS_FCT=0)
target_include_directories(openphysical_nfiq2 BEFORE PRIVATE "${OPENCV_OVERLAY}")
target_include_directories(openphysical_nfiq2 SYSTEM PRIVATE "${A}/include" "${A}/src" "${A}/src/nfiq2" "${FRFXLL_PATH}/include" "${U}/digestpp" "${U}/opencv/modules/core/include" "${U}/opencv/modules/imgproc/include" "${U}/opencv/modules/ml/include" "${CMAKE_CURRENT_BINARY_DIR}/opencv" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(openphysical_nfiq2 PRIVATE FRFXLL opencv_core opencv_imgproc opencv_ml)
if(APPLE)
 set_target_properties(openphysical_nfiq2 FRFXLL PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
 set_target_properties(openphysical_nfiq2 FRFXLL PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
if(WIN32)
 set_target_properties(openphysical_nfiq2 FRFXLL PROPERTIES PREFIX "")
 target_compile_definitions(openphysical_nfiq2 PRIVATE NOMINMAX)
endif()
set_source_files_properties(op_nfiq2.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
install(TARGETS openphysical_nfiq2 FRFXLL opencv_core opencv_imgproc opencv_ml RUNTIME DESTINATION . LIBRARY DESTINATION .)
include(CTest)
if(BUILD_TESTING)
 target_compile_definitions(openphysical_nfiq2 PRIVATE OP_NFIQ2_TESTING)
 add_executable(nfiq2_tests ../../tests/native/test_nfiq2.c ../../tests/native/munit/munit.c)
 target_include_directories(nfiq2_tests PRIVATE . ../../tests/native/munit)
 target_compile_definitions(nfiq2_tests PRIVATE NFIQ_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures")
 target_link_libraries(nfiq2_tests PRIVATE openphysical_nfiq2)
 if(WIN32)
  add_test(NAME nfiq2_tests COMMAND nfiq2_tests)
 else()
  add_test(NAME nfiq2_tests COMMAND nfiq2_tests --no-fork)
 endif()
endif()
